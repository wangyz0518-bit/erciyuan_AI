cmake_minimum_required(VERSION 3.22)
project(live2d_avatar LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../.." ABSOLUTE)
set(SDK "${REPO}/.local/live2d/sdk/CubismSdkForNative-5-r.5")
if(NOT EXISTS "${SDK}/Core/include/Live2DCubismCore.h")
  message(FATAL_ERROR "Run tool/live2d.ps1 -Action prepare first")
endif()
add_library(Live2DCubismCore STATIC IMPORTED)
set_target_properties(Live2DCubismCore PROPERTIES
  IMPORTED_LOCATION "${SDK}/Core/lib/android/${ANDROID_ABI}/libLive2DCubismCore.a"
  INTERFACE_INCLUDE_DIRECTORIES "${SDK}/Core/include")
set(FRAMEWORK_SOURCE OpenGL)
add_subdirectory("${SDK}/Framework" "${CMAKE_CURRENT_BINARY_DIR}/Framework")
target_compile_definitions(Framework PUBLIC CSM_TARGET_ANDROID_ES2)
add_library(live2d_avatar SHARED avatar.cpp)
target_include_directories(live2d_avatar PRIVATE "${SDK}/Samples/OpenGL/thirdParty/stb")
target_link_libraries(live2d_avatar Framework Live2DCubismCore EGL GLESv2 android log)
target_link_options(live2d_avatar PRIVATE "-Wl,-z,max-page-size=16384")
